function getFilteredRecords() { const search = clean( $("search").value ); const priority = clean( $("priority-filter").value ); const department = clean( $("department-filter").value ); const origin = clean( $("origin-filter").value ); /* * V16 * * "Toutes" = contributions À faire. * * Les filtres rapides utilisent toutes les contributions * afin d'être strictement cohérents avec les KPI. */ let records = [...allRecords]; if ( quickFilter === "all" ) { records = records.filter( isTodo ); } else if ( quickFilter === "late" ) { records = records.filter( isLate ); } else if ( quickFilter === "close" ) { records = records.filter( isClose ); } else if ( quickFilter === "watch" ) { records = records.filter( isWatch ); } else if ( quickFilter === "remind" ) { records = records.filter( needsReminder ); } else if ( quickFilter === "urgent" ) { records = records.filter( isUrgent ); } else if ( quickFilter === "week" ) { records = records.filter( isThisWeek ); } records = records.filter( record => { if (!search) { return true; } const haystack = [ record.Numero, record.Titre, record.Description, record.Demandeur, record.Contributeur, record.Bureau_origine ] .map(clean) .join(" "); return haystack .includes( search ); } ); records = records.filter( record => { if (!priority) { return true; } return ( clean( record.Priorite ) === priority ); } ); records = records.filter( record => { if (!department) { return true; } return ( clean( record.Contributeur ) === department ); } ); records = records.filter( record => { if (!origin) { return true; } return ( clean( record.Bureau_origine ) === origin ); } ); records.sort( (a, b) => { const da = dateValue( a.Echeance ); const db = dateValue( b.Echeance ); if ( !da && !db ) { return 0; } if (!da) { return 1; } if (!db) { return -1; } return ( da.getTime() - db.getTime() ); } ); return records; }
